// Extrait : Mon calcul du taux d'occupation par tranche horaire dans Reservations.jsx
const getOccupiedTablesCount = (slotTime) => {
    if (!formData.date || !existingReservations.length) return 0;

    const slotDateTime = new Date(`${formData.date} ${slotTime}:00`);

    return existingReservations.filter(res => {
        const start = new Date(res.start_time);
        const end = new Date(res.end_time);
        // Je vérifie si l'heure analysée est comprise dans une réservation existante
        return slotDateTime >= start && slotDateTime < end;

5. Sécurité JWT, Hashage Bcrypt & Middlewares RBAC : auth.controller.js & auth.middleware.js AT2 : Sécurité & OWASP

• Mon rôle & objectif métier : J'ai mis en place l'authentification sécurisée des utilisateurs, la délivrance des jetons signés cryptographiquement et la protection des points d'accès de mon API REST.

• Pourquoi je présente ce fichier au jury : J'y applique strictement les recommandations du référentiel DWWM et de l'OWASP : chiffrement irréversible des mots de passe, tokens temporisés et contrôle d'accès basé sur les rôles (RBAC).

Fonctionnement clé de mon code :

  • Hashage Bcrypt : Je sale et je hache les mots de passe avec un coût de 10 avant tout enregistrement en base SQL. Aucun mot de passe en clair n'est stocké.
  • Signature numérique JWT : Lors de la connexion, je délivre un jeton JWT contenant l'identifiant et le rôle de l'utilisateur, signé avec une clé secrète et valide 24 heures.
  • Pipeline Middleware (authMiddleware) : J'extrais le jeton dans l'en-tête Authorization: Bearer, je valide cryptographiquement la signature via jwt.verify() et j'injecte l'utilisateur dans l'objet requête req.user.
  • Contrôle RBAC (adminMiddleware) : Je vérifie la présence du rôle ADMIN et je bloque toute tentative non autorisée avec un statut HTTP standard 403 Forbidden.
// Extrait : Mon middleware d'authentification JWT (backend/src/middlewares/auth.middleware.js)
const authMiddleware = async (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;
        if (!authHeader || !authHeader.startsWith('Bearer ')) {
            return res.status(401).json({ error: 'Token manquant' });
        }

        const token = authHeader.split(' ')[1];
        const decoded = jwt.verify(token, process.env.JWT_SECRET);
        const user = await User.findById(decoded.id);

        if (!user) return res.status(401).json({ error: 'Utilisateur non trouvé' });

        req.user = user; // J'injecte l'utilisateur validé dans la requête
        next();
    } catch (error) {
        return res.status(401).json({ error: 'Token invalide ou expiré' });
    }
};

• Pourquoi je présente ce fichier au jury : Il prouve ma maîtrise de l'intégrité relationnelle SQL (clés composites) et ma gestion des exceptions natives du moteur MySQL (ER_DUP_ENTRY).

Fonctionnement clé de mon code :

  • Contrôle des jauges de capacité : Je vérifie avant toute écriture que registeredCount < capacity, et je renvoie un code HTTP 400 explicite si le quota est atteint.
  • Clé composite & Interception d'erreur SQL : Ma table tournament_participants s'appuie sur une clé primaire composite (tournament_id, user_id). Si un utilisateur tente de s'inscrire deux fois, le moteur MySQL génère l'erreur ER_DUP_ENTRY que j'intercepte dans le bloc catch pour informer proprement le client sans planter le serveur.
// Extrait : Mon inscription sécurisée avec capture du doublon SQL (tournament.controller.js)
export const registerForTournament = async (req, res) => {
    try {
        const tournamentId = req.params.id;
        const userId = req.user.id;

        const tourneyInfo = await Tournament.getCapacityAndCount(tournamentId);
        if (tourneyInfo.registeredCount >= tourneyInfo.capacity) {
            return res.status(400).json({ error: "Ce tournoi est complet." });
        }

        await Tournament.register(tournamentId, userId);
        res.status(201).json({ message: "Inscription réussie !" });
    } catch (error) {
        if (error.code === 'ER_DUP_ENTRY') {
            return res.status(400).json({ error: "Vous êtes déjà inscrit à ce tournoi." });
        }
        res.status(500).json({ error: "Erreur serveur lors de l'inscription." });
    }
};
